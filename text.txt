const API_ENDPOINTS = [
  "https://api.binance.com/api/v3/klines",
  "https://api1.binance.com/api/v3/klines",
  "https://api2.binance.com/api/v3/klines",
  "https://api3.binance.com/api/v3/klines"
];

const SYMBOL = "BTCUSDT";
const INTERVAL = "15m";
const LIMIT = 150;

let currentEndpoint = 0;
let busy = false;
let lastSuccessfulUpdate = 0;

const $ = id => document.getElementById(id);

function setText(id, value){
  const el = $(id);
  if(el && el.textContent !== String(value)){
    el.textContent = String(value);
  }
}

function money(value){
  if(!Number.isFinite(value)) return "$--";
  return "$" + value.toLocaleString("en-US",{
    minimumFractionDigits:2,
    maximumFractionDigits:2
  });
}

function number(value, digits=2){
  if(!Number.isFinite(value)) return "--";
  return value.toLocaleString("en-US",{
    minimumFractionDigits:digits,
    maximumFractionDigits:digits
  });
}

function percent(value){
  if(!Number.isFinite(value)) return "--";
  return (value >= 0 ? "+" : "") +
    value.toFixed(3) + "%";
}

function setConnection(ok, message){

  const dot = $("connectionDot");

  dot.className = "dot " + (ok ? "ok" : "bad");

  setText(
    "connectionText",
    message
  );
}

function formatTime(ms){

  const d = new Date(ms);

  return d.toLocaleTimeString("en-US",{
    hour12:false,
    hour:"2-digit",
    minute:"2-digit",
    second:"2-digit"
  });
}

function formatCountdown(ms){

  if(!Number.isFinite(ms) || ms < 0){
    return "--:--";
  }

  const total = Math.floor(ms / 1000);

  const minutes = Math.floor(total / 60);
  const seconds = total % 60;

  return String(minutes).padStart(2,"0") +
    ":" +
    String(seconds).padStart(2,"0");
}

/* ============================================================
   INDICADORES
   ============================================================ */

function ema(values, period){

  if(values.length < period){
    return null;
  }

  const multiplier = 2 / (period + 1);

  let result = 0;

  for(let i=0;i<period;i++){
    result += values[i];
  }

  result /= period;

  for(let i=period;i<values.length;i++){
    result =
      (values[i] - result) * multiplier +
      result;
  }

  return result;
}

function calculateRSI(values, period=14){

  if(values.length <= period){
    return null;
  }

  let gain = 0;
  let loss = 0;

  for(let i=1;i<=period;i++){

    const change =
      values[i] - values[i-1];

    if(change >= 0){
      gain += change;
    }else{
      loss += Math.abs(change);
    }
  }

  gain /= period;
  loss /= period;

  for(let i=period+1;i<values.length;i++){

    const change =
      values[i] - values[i-1];

    const g =
      change > 0 ? change : 0;

    const l =
      change < 0 ? Math.abs(change) : 0;

    gain =
      ((gain * (period - 1)) + g) / period;

    loss =
      ((loss * (period - 1)) + l) / period;
  }

  if(loss === 0){
    return 100;
  }

  const rs = gain / loss;

  return 100 - (100 / (1 + rs));
}

function calculateVWAP(candles){

  let cumulativePV = 0;
  let cumulativeVolume = 0;

  for(const c of candles){

    const high = Number(c[2]);
    const low = Number(c[3]);
    const close = Number(c[4]);
    const volume = Number(c[5]);

    const typical =
      (high + low + close) / 3;

    cumulativePV += typical * volume;
    cumulativeVolume += volume;
  }

  if(cumulativeVolume === 0){
    return null;
  }

  return cumulativePV / cumulativeVolume;
}

function calculateMomentum(closes){

  if(closes.length < 10){
    return null;
  }

  const current =
    closes[closes.length - 1];

  const previous =
    closes[closes.length - 10];

  if(previous === 0){
    return null;
  }

  return ((current - previous) / previous) * 100;
}

/* ============================================================
   OBTENER DATOS DE BINANCE
   ============================================================ */

async function fetchKlines(){

  let attempts = 0;

  while(attempts < API_ENDPOINTS.length){

    const endpoint =
      API_ENDPOINTS[currentEndpoint];

    try{

      const controller =
        new AbortController();

      const timeout =
        setTimeout(
          () => controller.abort(),
          8000
        );

      const url =
        endpoint +
        "?symbol=" +
        SYMBOL +
        "&interval=" +
        INTERVAL +
        "&limit=" +
        LIMIT;

      const response =
        await fetch(url,{
          method:"GET",
          cache:"no-store",
          signal:controller.signal,
          headers:{
            "Accept":"application/json"
          }
        });

      clearTimeout(timeout);

      if(!response.ok){
        throw new Error(
          "HTTP " + response.status
        );
      }

      const data =
        await response.json();

      if(!Array.isArray(data) || data.length < 30){
        throw new Error("Datos insuficientes");
      }

      lastSuccessfulUpdate =
        Date.now();

      return data;

    }catch(error){

      currentEndpoint =
        (currentEndpoint + 1) %
        API_ENDPOINTS.length;

      attempts++;
    }
  }

  throw new Error(
    "No fue posible conectar con Binance"
  );
}

/* ============================================================
   SEÑAL V5
   ============================================================ */

function calculateSignal(data){

  const closes =
    data.map(c => Number(c[4]));

  const current =
    closes[closes.length - 1];

  const previous =
    closes[closes.length - 2];

  const ema9 =
    ema(closes,9);

  const ema21 =
    ema(closes,21);

  const rsi =
    calculateRSI(closes,14);

  const vwap =
    calculateVWAP(data);

  const momentum =
    calculateMomentum(closes);

  let bull = 0;
  let bear = 0;

  /* EMA */

  if(ema9 > ema21){
    bull += 2;
  }else{
    bear += 2;
  }

  /* Precio contra EMA */

  if(current > ema9){
    bull += 1;
  }else{
    bear += 1;
  }

  /* VWAP */

  if(current > vwap){
    bull += 1;
  }else{
    bear += 1;
  }

  /* RSI */

  if(rsi >= 55){
    bull += 2;
  }else if(rsi <= 45){
    bear += 2;
  }

  /* Momentum */

  if(momentum > 0.05){
    bull += 2;
  }else if(momentum < -0.05){
    bear += 2;
  }

  const total =
    bull + bear;

  let signal = "WAIT";

  if(
    bull >= 6 &&
    bull >= bear + 2 &&
    rsi < 72
  ){
    signal = "UP";
  }
  else if(
    bear >= 6 &&
    bear >= bull + 2 &&
    rsi > 28
  ){
    signal = "DOWN";
  }

  const strongest =
    Math.max(bull,bear);

  let confidence =
    total > 0
      ? (strongest / total) * 100
      : 0;

  /*
    Evitamos mostrar una falsa precisión.
    El modelo limita la confianza.
  */

  confidence =
    Math.max(
      50,
      Math.min(95,confidence)
    );

  return {
    current,
    ema9,
    ema21,
    rsi,
    vwap,
    momentum,
    bull,
    bear,
    signal,
    confidence
  };
}

/* ============================================================
   ENTRADA / SL / TP
   ============================================================ */

function calculateTrade(signalData, candle){

  const price =
    signalData.current;

  const high =
    Number(candle[2]);

  const low =
    Number(candle[3]);

  let entry = price;
  let sl = null;
  let tp = null;

  if(signalData.signal === "UP"){

    const risk =
      Math.max(
        price * 0.0015,
        price - low
      );

    sl =
      price - risk;

    tp =
      price + risk * 1.5;
  }

  if(signalData.signal === "DOWN"){

    const risk =
      Math.max(
        price * 0.0015,
        high - price
      );

    sl =
      price + risk;

    tp =
      price - risk * 1.5;
  }

  return {
    entry,
    sl,
    tp
  };
}

      "signalSub",

  }
  else if(signalData.signal === "DOWN"){

    main.classList.add("red");

    setText(
      "signalMain",

    );

      "signalSub",

      "signalMain",

        : "NEUTRAL"

      : "$--"

  }
}

/* ============================================================
   ACTUALIZACIÓN PRINCIPAL
   ============================================================ */

async function updateBot(){

  if(busy){
    return;
  }

  busy = true;

  try{

    setConnection(
      true,
      "Conectado · actualizando datos..."
    );

    const data =
      await fetchKlines();

    updateUI(data);

    setConnection(
      true,
      "🟢 Binance conectado · actualización cada 5 s"
    );

  }catch(error){

    console.error(error);

    /*
      No borramos los datos anteriores.
      Esto evita que la pantalla salte a cero
      durante una falla momentánea de red.
    */

    setConnection(
      false,
      "🔴 Sin conexión · intentando reconectar..."
    );

  }finally{

    busy = false;
  }
}

    La cuenta regresiva se actualiza localmente

    Date.now();

  const remainder =
    15 * 60 * 1000 -
    (now % (15 * 60 * 1000));

  setText(
    "countdown",
    formatCountdown(remainder)
  );

},1000);

/* ============================================================
   ACTUALIZACIÓN REAL CADA 5 SEGUNDOS
   ============================================================ */

updateBot();

setInterval(
  updateBot,
