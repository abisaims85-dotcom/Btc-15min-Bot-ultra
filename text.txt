<script>
(()=>{'use strict';
const $=id=>document.getElementById(id);
const state={candles:[],price:NaN,source:'',last:0,busy:false,history:[]};
const INTERVAL=900000, UPDATE_MS=5000, MAX_AGE=20000;
const fmt=n=>Number.isFinite(n)?'$'+n.toLocaleString('en-US',{minimumFractionDigits:2,maximumFractionDigits:2}):'—';
const num=(n,d=2)=>Number.isFinite(n)?n.toFixed(d):'—';
const put=(id,v)=>$(id).textContent=v;

function ema(vals,p){
 if(vals.length<p)return NaN;
 const k=2/(p+1);let e=vals[0];
 for(let i=1;i<vals.length;i++)e=vals[i]*k+e*(1-k);
 return e;
}
function rsi(vals,p=14){
 if(vals.length<p+1)return NaN;
 let g=0,l=0;
 for(let i=vals.length-p;i<vals.length;i++){
  const d=vals[i]-vals[i-1];
  if(d>0)g+=d;else l-=d;
 }
 if(l===0)return g===0?50:100;
 if(g===0)return 0;
 return 100-100/(1+g/l);
}
function atr(cs,p=14){
 if(cs.length<p+1)return NaN;
 const a=[];
 for(let i=cs.length-p;i<cs.length;i++){
  const c=cs[i],pr=cs[i-1];
  a.push(Math.max(c.high-c.low,Math.abs(c.high-pr.close),Math.abs(c.low-pr.close)));
 }
 return a.reduce((x,y)=>x+y,0)/a.length;
}
function vwap(cs){
 const s=cs.slice(-20);let pv=0,vv=0;
 for(const c of s){
  const v=Number.isFinite(c.volume)&&c.volume>0?c.volume:1;
  pv+=(c.high+c.low+c.close)/3*v;vv+=v;
 }
 return vv?pv/vv:NaN;
}
async function json(url,ms=6500){
 const ctl=new AbortController();
 const timer=setTimeout(()=>ctl.abort(),ms);
 try{
  const r=await fetch(url,{cache:'no-store',signal:ctl.signal,headers:{Accept:'application/json'}});
  if(!r.ok)throw Error('HTTP '+r.status);
  return await r.json();
 }finally{clearTimeout(timer)}
}
function parseBinance(d){
 if(!Array.isArray(d)||d.length<25)throw Error('Historial Binance insuficiente');
 return d.map(a=>({
  time:+a[0],open:+a[1],high:+a[2],low:+a[3],close:+a[4],volume:+a[5]
 })).filter(c=>[c.time,c.open,c.high,c.low,c.close,c.volume].every(Number.isFinite));
}
async function binance(){
 const [d,t]=await Promise.all([
  json('https://api.binance.com/api/v3/klines?symbol=BTCUSDT&interval=15m&limit=100'),
  json('https://api.binance.com/api/v3/ticker/price?symbol=BTCUSDT')
 ]);
 const cs=parseBinance(d),p=+t.price;
 if(!Number.isFinite(p)||p<=0)throw Error('Precio Binance inválido');
 return{candles:cs,price:p,source:'Binance BTC/USDT'};
}
async function coinbase(){
 const end=Date.now(),start=end-36*60*60*1000;
 const base='https://api.exchange.coinbase.com/products/BTC-USD';
 const qs='?granularity=900&start='+encodeURIComponent(new Date(start).toISOString())+'&end='+encodeURIComponent(new Date(end).toISOString());
 const [d,t]=await Promise.all([
  json(base+'/candles'+qs),
  json(base+'/ticker')
 ]);
 if(!Array.isArray(d)||d.length<25)throw Error('Historial Coinbase insuficiente');
 // Coinbase Exchange: [segundos, mínimo, máximo, apertura, cierre, volumen]
 const cs=d.map(a=>({
  time:+a[0]*1000,low:+a[1],high:+a[2],open:+a[3],close:+a[4],volume:+a[5]
 })).filter(c=>[c.time,c.open,c.high,c.low,c.close,c.volume].every(Number.isFinite))
 .sort((a,b)=>a.time-b.time);
 const p=+t.price;
 if(!Number.isFinite(p)||p<=0)throw Error('Precio Coinbase inválido');
 return{candles:cs,price:p,source:'Coinbase BTC/USD'};
}
function connection(s,color){
 put('connection',s);$('connection').style.color=color||'var(--muted)';
}
function clearMarket(msg){
 put('lamp','🟡');put('signal','ESPERAR');
 $('signal').className='yellow';put('reason',msg);
 ['price','estimate','open','range','momentum','ema9','ema21','rsi','vwap','atr','confidence','entry','stop','target'].forEach(id=>put(id,'—'));
 put('score','Sin datos actuales verificados');
 put('levelsNote','Sin datos frescos: niveles ocultos.');
}
async function update(){
 if(state.busy)return;
 state.busy=true;
 const errors=[];
 try{
  let data=null;
  // Fuente principal y respaldo. No se mezclan velas de distintas fuentes.
  for(const fn of [binance,coinbase]){
   try{
    const d=await fn(),now=Date.now();
    if(!Array.isArray(d.candles)||d.candles.length<25)throw Error('Velas insuficientes');
    const cs=d.candles.map(c=>({...c})).sort((a,b)=>a.time-b.time);
    const candleStart=Math.floor(now/INTERVAL)*INTERVAL;
    const latest=cs[cs.length-1];
    if(!Number.isFinite(latest.time)||now-latest.time>INTERVAL+60000)
     throw Error('Velas atrasadas');
    let cur=cs.find(x=>x.time===candleStart);
    if(!cur){
     if(latest.time<candleStart)throw Error('La fuente aún no publicó la vela actual');
     cur=latest;
    }
    // El precio en vivo actualiza la vela actual, no las velas anteriores.
    cur.close=d.price;
    cur.high=Math.max(cur.high,d.price);
    cur.low=Math.min(cur.low,d.price);
    data={candles:cs,price:d.price,source:d.source,last:now};
    break;
   }catch(e){errors.push(fn.name+': '+(e.message||'error'))}
  }
  if(!data){
   connection('SIN DATOS · REINTENTANDO','var(--yellow)');
   put('sourceLine','Fallaron las fuentes: '+errors.join(' | '));
   clearMarket('No hay datos actuales verificables. Reintentando automáticamente.');
   return;
  }
  state.candles=data.candles;
  state.price=data.price;
  state.source=data.source;
  state.last=data.last;
  render();
 }catch(e){
  connection('ERROR · REINTENTANDO','var(--yellow)');
  put('sourceLine',e.message||'Error al procesar datos.');
  clearMarket('Error al procesar los datos. Reintentando.');
 }finally{
  state.busy=false;
 }
}
function render(){
 const cs=state.candles,p=state.price,now=Date.now();
 if(!cs.length||!Number.isFinite(p)||now-state.last>MAX_AGE){
  connection('DATOS VENCIDOS','var(--yellow)');
  clearMarket('Los datos no están actualizados. Espera la siguiente conexión.');
  return;
 }
 const c=cs[cs.length-1],cl=cs.map(x=>x.close);
 const e9=ema(cl,9),e21=ema(cl,21),rr=rsi(cl),aa=atr(cs),vv=vwap(cs);
 const mom=cs.length>=4?(p-cs[cs.length-4].close)/cs[cs.length-4].close*100:NaN;
 const prior=cs[cs.length-2],prev3=cs[cs.length-3];
 const left=Math.max(0,(Math.floor(now/INTERVAL)+1)*INTERVAL-now);

 put('price',fmt(p));put('estimate',fmt(p));put('open',fmt(c.open));
 put('range',fmt(c.high)+' / '+fmt(c.low));
 put('momentum',(mom>0?'+':'')+num(mom,3)+'%');
 put('ema9',fmt(e9));put('ema21',fmt(e21));put('rsi',num(rr,1));
 put('vwap',fmt(vv));put('atr',fmt(aa));
 put('countdown',String(Math.floor(left/60000)).padStart(2,'0')+':'+String(Math.floor(left%60000/1000)).padStart(2,'0'));
 put('updated',new Date(state.last).toLocaleTimeString());
 connection('CONECTADO · '+state.source,'var(--green)');
 put('sourceLine','Datos recibidos. Revisión automática cada 5 segundos.');

 if(![e9,e21,rr,aa,vv,mom].every(Number.isFinite)){
  clearMarket('Reuniendo datos suficientes para calcular indicadores.');
  return;
 }

 let bull=0,bear=0;
 const vote=(up,down)=>{if(up)bull++;else if(down)bear++;};
 vote(e9>e21,e9<e21);
 vote(p>e9,p<e9);
 vote(p>vv,p<vv);
 vote(rr>=52,rr<=48);
 vote(mom>=0.025,mom<=-0.025);
 vote(p>c.open,p<c.open);
 vote(!!(prior&&prev3&&p>prior.close&&prior.close>prev3.close),
      !!(prior&&prev3&&p<prior.close&&prior.close<prev3.close));

 // Señales rápidas de pérdida de fuerza: primero ESPERAR, no BAJA automática.
 const weakening=
  p<e9 ||
  mom<=-0.015 ||
  rr<48 ||
  (p<c.open&&prior&&p<prior.close);

 const strongUp=
  bull>=5&&bull-bear>=3&&p>e9&&mom>0.015&&rr>=50&&!weakening;

 const strongDown=
  bear>=5&&bear-bull>=3&&p<e9&&mom< -0.015&&rr<=50;

 let sig='ESPERAR',reason='Las señales no coinciden lo suficiente. Espera.';
 let cls='yellow',lamp='🟡';
 let conf=Math.round(Math.max(bull,bear)/7*100);

 if(strongDown){
  sig='BAJA';reason='Presión bajista confirmada por varios indicadores.';
  cls='red';lamp='🔴';
 }else if(weakening){
  reason='El precio pierde fuerza. Se pausa SUBE antes de una posible caída.';
  conf=Math.max(conf,45);
 }else if(strongUp){
  sig='SUBE';reason='Varias señales alcistas coinciden y el precio conserva fuerza.';
  cls='green';lamp='🟢';
 }else{
  reason='Todavía no hay confirmación suficiente para SUBE o BAJA.';
 }

 put('signal',sig);$('signal').className=cls;
 put('lamp',lamp);put('reason',reason);
 put('score','Votos alcistas: '+bull+' · bajistas: '+bear);
 put('confidence',conf+'%');

 if(sig!=='ESPERAR'&&aa>0){
  put('entry',fmt(p));
  put('stop',fmt(sig==='SUBE'?p-1.2*aa:p+1.2*aa));
  put('target',fmt(sig==='SUBE'?p+1.8*aa:p-1.8*aa));
  put('levelsNote','Niveles ilustrativos basados en ATR; no son órdenes.');
 }else{
  ['entry','stop','target'].forEach(id=>put(id,'—'));
  put('levelsNote','Sin señal direccional confirmada; niveles ocultos.');
 }
 addHistory({time:new Date().toLocaleTimeString(),source:state.source,price:p,sig});
}
function addHistory(h){
 const last=state.history[state.history.length-1];
 if(last&&last.source===h.source&&last.sig===h.sig&&Math.abs(last.price-h.price)<1)return;
 state.history.push(h);state.history=state.history.slice(-8);
 const body=$('history');body.innerHTML='';
 [...state.history].reverse().forEach(h=>{
  const tr=document.createElement('tr');
  [h.time,h.source,fmt(h.price),h.sig].forEach((v,i)=>{
   const td=document.createElement('td');td.textContent=v;
   if(i===3)td.style.color=h.sig==='SUBE'?'var(--green)':h.sig==='BAJA'?'var(--red)':'var(--yellow)';
   tr.appendChild(td);
  });
  body.appendChild(tr);
 });
}
$('refresh').addEventListener('click',update);
$('clear').addEventListener('click',()=>{
 state.history=[];
 $('history').innerHTML='<tr><td colspan="4">Historial borrado.</td></tr>';
});
update();
setInterval(update,UPDATE_MS);
setInterval(()=>{
 const left=Math.max(0,(Math.floor(Date.now()/INTERVAL)+1)*INTERVAL-Date.now());
 put('countdown',String(Math.floor(left/60000)).padStart(2,'0')+':'+String(Math.floor(left%60000/1000)).padStart(2,'0'));
 if(state.last&&Date.now()-state.last>MAX_AGE){
  connection('DATOS VENCIDOS','var(--yellow)');
  clearMarket('Los datos dejaron de actualizarse. No uses la señal anterior.');
 }
},1000);
})();
