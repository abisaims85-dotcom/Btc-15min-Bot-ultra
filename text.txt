async function fetchFresh(url){

  const controller = new AbortController();

  const timer = setTimeout(
    () => controller.abort(),
    REQUEST_TIMEOUT
  );

  try{

    const separator =
      url.includes("?") ? "&" : "?";

    const finalURL =
      url +
      separator +
      "_v52=" +
      Date.now();

    const response = await fetch(
      finalURL,
      {
        method:"GET",
        cache:"no-store",
        signal:controller.signal,
        headers:{
          "Cache-Control":"no-cache",
          "Pragma":"no-cache"
        }
      }
    );

    if(!response.ok){

      throw new Error(
        "HTTP " + response.status
      );

    }

    return await response.json();

  }finally{

    clearTimeout(timer);

  }

}

  const priceResult = results[1];

  let candleData =
    candleResult.status === "fulfilled"
      ? candleResult.value
      : null;

  let priceData =
    priceResult.status === "fulfilled"
      ? priceResult.value
      : null;


  /*
     Si no tenemos ticker pero sí velas,
     usamos el último close fresco.
  */

  if(!priceData && candleData){

    const last =
      candleData.candles[
        candleData.candles.length-1
      ];

    if(
      Number.isFinite(last.close) &&
      last.close > 0
    ){

      priceData = {

        price:last.close,

        source:candleData.source,

        receivedAt:Date.now(),

        fallback:true

      };

    }

  }


  if(!candleData || !priceData){

    throw new Error(
      "No hay datos suficientes"
    );

  }


  const candles =
    candleData.candles.slice();


  const price =
    priceData.price;

  const now = Date.now();

  const slot =
    Math.floor(
      now / (15*60*1000)
    ) * 15*60*1000;


  let current =
    candles[candles.length-1];


  /*
     Si Binance ya tiene la vela actual,
     actualizarla con el precio actual.
  */

  if(current.openTime === slot){

    current = {
      ...current,

      close:price,

      high:Math.max(
        current.high,
        price
      ),

      low:Math.min(
        current.low,
        price
      ),

      closeTime:
        slot + 15*60*1000 - 1

    };

    candles[candles.length-1] =
      current;

  }

  /*
     Si la última vela es anterior,
     crear solamente la vela actual.
  */

  else if(current.openTime < slot){

    const previous = current;

    current = {

      openTime:slot,

      open:price,

      high:price,

      low:price,

      close:price,

      volume:0,

      closeTime:
        slot + 15*60*1000 - 1

    };

    candles.push(current);

  }


  return {

    candles,

    price,

    source:
      priceData.source +
      (priceData.fallback ? " · respaldo" : ""),

    receivedAt:Math.max(
      candleData.receivedAt,
      priceData.receivedAt
    ),

    candle:current

  };

}


/* =========================================================

function ema(values,period){

  if(values.length < period){

    return null;

  }

  const k =
    2/(period+1);

  let e =
    values
      .slice(0,period)
      .reduce(
        (a,b)=>a+b,
        0
      ) / period;

  for(
    let i=period;
    i<values.length;
    i++
  ){

    e =
      values[i] * k +
      e * (1-k);

  }

  return e;

}


function rsi(values,period=14){

  if(values.length <= period){

    return null;

  }

  let gain=0;
  let loss=0;

  for(
    let i=1;
    i<=period;
    i++
  ){

    const d =
      values[i]-values[i-1];

    if(d>=0) gain+=d;
    else loss-=d;

  }

  gain/=period;
  loss/=period;

  for(
    let i=period+1;
    i<values.length;
    i++
  ){

    const d =
      values[i]-values[i-1];

    gain =
      (
        gain*(period-1) +
        Math.max(0,d)
      ) / period;

    loss =
      (
        loss*(period-1) +
        Math.max(0,-d)
      ) / period;

  }

  if(loss===0){

    return 100;

  }

  return 100 -
    100/(1+gain/loss);

}


function atr(data,period=14){

  if(data.length < period+1){

    return null;

  }

  const tr=[];

  for(
    let i=1;
    i<data.length;
    i++
  ){

    const c=data[i];
    const previous=data[i-1].close;

    tr.push(
      Math.max(
        c.high-c.low,

        Math.abs(
          c.high-previous
        ),

        Math.abs(
          c.low-previous
        )
      )
    );

  }

  let value =
    tr
      .slice(0,period)
      .reduce(
        (a,b)=>a+b,
        0
      ) / period;

  for(
    let i=period;
    i<tr.length;
    i++
  ){

    value =
      (
        value*(period-1) +
        tr[i]
      ) / period;

  }

  return value;

}


function vwap(data){

  let pv=0;
  let volume=0;

  for(const c of data){

    const typical =
      (
        c.high +
        c.low +
        c.close
      ) / 3;

    const vol =
      Number.isFinite(c.volume)
        ? c.volume
        : 0;

    pv += typical*vol;

    volume += vol;

  }

  return volume>0
    ? pv/volume
    : null;

}


function pct(a,b){

  if(
    !Number.isFinite(a) ||
    !Number.isFinite(b) ||
    b===0
  ){

    return 0;

  }

  return (
    (a-b)/b
  )*100;

}

  if(slope>.03){

    score+=1.5;

  }else if(slope<-.03){

    score-=1.5;

  }


  if(r>55){

    score+=1;

  }else if(r<45){

    score-=1;

  }


  if(body>.08 && position>.62){

    score+=1;

  }


  if(body<-.08 && position<.38){

    score-=1;

  }


  if(r>76){

    score-=1.5;

  }


  if(r<24){

    score+=1.5;

  }


  let raw =
    score>=2.2
      ? "UP"
      : score<=-2.2
        ? "DOWN"
        : "WAIT";


  /*
     Protección contra cambios demasiado rápidos.
  */

  let invalid=false;


  if(lastSignal==="UP"){

    invalid =
      (
        score<0.5
      ) ||
      (
        e9 &&
        price<e9 &&
        momentum<-.08
      );

  }


  if(lastSignal==="DOWN"){

    invalid =
      (
        score>-0.5
      ) ||
      (
        e9 &&
        price>e9 &&
        momentum>.08
      );

  }


  if(
    lastSignal!=="WAIT" &&
    raw!==lastSignal &&
    !invalid
  ){

    raw="WAIT";

  }


  /*
     Se requieren dos lecturas consecutivas
     para confirmar un cambio.
  */

  if(raw!==lastSignal){

    if(pendingSignal===raw){

      pendingCount++;

    }else{

      pendingSignal=raw;
      pendingCount=1;

    }


    if(pendingCount>=2){

      lastSignal=raw;

      pendingSignal=null;
      pendingCount=0;

    }

  }else{

    pendingSignal=null;
    pendingCount=0;

  }


  /*
     Si la señal actual queda invalidada,
     pasa inmediatamente a ESPERAR.
  */

  if(
    invalid &&
    raw!==lastSignal
  ){

    lastSignal="WAIT";

    pendingSignal=null;
    pendingCount=0;

  }


  const confidence =
    Math.min(
      99,
      Math.round(
        50 +
        Math.min(
          42,
          Math.abs(score)*7
        )
      )
    );


  let reason =
    "Fuerzas mezcladas";


  if(lastSignal==="UP"){

    reason =
      "Estructura alcista · score " +
      score.toFixed(1);

  }else if(lastSignal==="DOWN"){

    reason =
      "Estructura bajista · score " +
      score.toFixed(1);

  }else{

    reason =
      invalid
        ? "Señal invalidada · esperar confirmación"
        : "Sin confirmación suficiente para entrar";

  }


  return {

    current,

    price,

    e9,

    e21,

    r,

    vw,

    a,

    momentum,

    score,

    signal:lastSignal,

    confidence,

    reason,

    receivedAt:Date.now()

  };

}


/* =========================================================
   ENTRADA / SL / TP
   ========================================================= */

function updateLevels(a){

  if(
    a.signal==="WAIT" ||
    !Number.isFinite(a.a)
  ){

    currentLevels={
      entry:null,
      sl:null,
      tp:null
    };

  }else{

    const risk =
      Math.max(
        a.a,
        a.price*0.0015
      );


    currentLevels={

      entry:a.price,

      sl:
        a.signal==="UP"
          ? a.price-risk*1.05
          : a.price+risk*1.05,

      tp:
        a.signal==="UP"
          ? a.price+risk*1.55
          : a.price-risk*1.55

    };

  }


  setText(
    "entry",
    currentLevels.entry
      ? "$"+fmt(currentLevels.entry)
      : "—"
  );


  setText(
    "sl",
    currentLevels.sl
      ? "$"+fmt(currentLevels.sl)
      : "—"
  );


  setText(
    "tp",
    currentLevels.tp
      ? "$"+fmt(currentLevels.tp)
      : "—"
  );

}

/* =========================================================
   REFRESH PRINCIPAL
   ========================================================= */

async function refresh(){

  if(refreshing){

    return;

  }


  refreshing=true;


  try{

    const live =
      await getLiveMarket();


    if(
      !Number.isFinite(live.price) ||
      live.price<=0
    ){

      throw new Error(
        "Precio inválido"
      );

    }


    if(
      Date.now()-live.receivedAt >
      MAX_AGE_MS
    ){

      throw new Error(
        "Lectura demasiado vieja"
      );

    }


    /*
       Si empieza una nueva vela,
       limpiamos el estado pendiente
       pero NO forzamos una señal.
    */

    if(
      lastCandleOpen &&
      live.candle.openTime !==
      lastCandleOpen
    ){

      pendingSignal=null;
      pendingCount=0;

    }


    lastCandleOpen =
      live.candle.openTime;


    const analysis =
      analyze(
        live.candles,
        live.price
      );


    lastLivePrice =
      live.price;


    lastSource =
      live.source;


    lastAnalysis =
      analysis;


    updateUI(
      analysis,
      live.source
    );


  }catch(error){

    /*
       IMPORTANTE:
       si ya tenemos una lectura fresca,
       NO borramos la pantalla por un
       fallo aislado de una petición.
    */

    if(
      lastAnalysis &&
      Date.now() -
      lastAnalysis.receivedAt <=
      MAX_AGE_MS
    ){

      setText(
        "connection",
        "● DATOS ANTERIORES · REINTENTANDO"
      );

      $("connection").className =
        "badge";

    }else{

      clearMarket();

    }

  }finally{

    refreshing=false;

  }

}
